        async function showPage(pageId) {

            document.querySelectorAll('.page-content').forEach(page => page.style.display = 'none');

            const pageElement = document.getElementById(`page-${pageId}`);

            pageElement.style.display = 'block';

            document.querySelectorAll('.nav-link').forEach(link => link.classList.remove('active'));

            document.querySelector(`a[onclick="showPage('${pageId}')"]`).classList.add('active');

            const subheading = document.getElementById('page-subheading');

            subheading.textContent = (pageId === 'rules') ? 'League Rules & Information' : `${pageId} Scores & Standings`;

            if (pageId !== 'rules' && pageElement.innerHTML.trim() === '') {

            }

        async function generateSeasonPage(year) {

            const container = document.getElementById(`page-${year}`);

            const data = seasonData[year];

            if (!data) return;

            container.innerHTML = `<div id="content-container-${year}"><div class="text-center text-gray-400 p-8">Loading ${year} season data...</div></div>`;

            if (data.isLive) {

                await fetchAndRenderLiveSeason(year);

            }

        }


        async function fetchAndRenderLiveSeason(year) {

            const API_KEY = '123';

            const NFL_LEAGUE_ID = '4391';

            const WEEKS = 18;

            let allEvents = [];

            const contentContainer = document.getElementById(`content-container-${year}`);


            for (let week = 1; week <= WEEKS; week++) {

                try {

                    const url = `https://www.thesportsdb.com/api/v1/json/${API_KEY}/eventsround.php?id=${NFL_LEAGUE_ID}&r=${week}&s=${year}`;

                    const response = await fetch(url);

                    const apiData = await response.json();

                    if (apiData.events) {

                        allEvents = allEvents.concat(apiData.events);

                    }

                    contentContainer.innerHTML = `<div class="text-center text-gray-400 p-8">Loading ${year} season data... Fetched week ${week}/${WEEKS}</div>`;

                    await new Promise(res => setTimeout(res, 1500)); 

                } catch (error) {

                    console.error(`Error fetching data for week ${week}:`, error);

                }

            }


            if (allEvents.length > 0) {

                const abbrToFullNameMap = {};

                allEvents.forEach(event => {

                    const homeTeamAbbr = TEAM_NAME_TO_ABBR_MAP[event.strHomeTeam] || event.strHomeTeam;

                    const awayTeamAbbr = TEAM_NAME_TO_ABBR_MAP[event.strAwayTeam] || event.strAwayTeam;

                    if (homeTeamAbbr !== event.strHomeTeam) abbrToFullNameMap[homeTeamAbbr] = event.strHomeTeam;

                    if (awayTeamAbbr !== event.strAwayTeam) abbrToFullNameMap[awayTeamAbbr] = event.strAwayTeam;

                });


                const processedGames = processApiData(allEvents);

                const liveData = {

                    ...seasonData[year],

                    games: processedGames,

                    abbrMap: abbrToFullNameMap,

                };

                renderTableAndCharts(year, liveData);

            } else {

                contentContainer.innerHTML = `<div class="text-center text-red-400 p-8">Could not fetch live data for ${year}. The API may be unavailable or the season has not started.</div>`;

            }

        }

        

        function processApiData(events) {

             return events.map(event => {

                const homeTeamAbbr = TEAM_NAME_TO_ABBR_MAP[event.strHomeTeam] || event.strHomeTeam;

                const awayTeamAbbr = TEAM_NAME_TO_ABBR_MAP[event.strAwayTeam] || event.strAwayTeam;

                return {

                    week: parseInt(event.intRound),

                    date: event.dateEvent,

                    home_team: homeTeamAbbr,

                    home_score: event.intHomeScore !== null ? parseInt(event.intHomeScore) : null,

                    away_team: awayTeamAbbr,

                    away_score: event.intAwayScore !== null ? parseInt(event.intAwayScore) : null,

                };

            });

        }


        function processStaticData(gameData, WEEKS) {

            const allGamesByWeek = new Map();

            const weekDateMap = {};

            for (let i = 1; i <= WEEKS; i++) allGamesByWeek.set(i, []);

            gameData.forEach(game => {

                const weekGames = allGamesByWeek.get(game.week);

                if (weekGames) weekGames.push(game);

                if (game.date && !weekDateMap[game.week]) {

                    weekDateMap[game.week] = game.date;

                }

            });

            return { allGamesByWeek, weekDateMap };

        }


        function renderTableAndCharts(year, overrideData = null) {

            const data = overrideData || seasonData[year];

            const { owners, games, abbrMap, currentWeek } = data;

            const WEEKS = 18; // Live season is always 18 weeks

            const { allGamesByWeek, weekDateMap } = processStaticData(games, WEEKS);


            const contentContainer = document.getElementById(`content-container-${year}`);

            contentContainer.innerHTML = `

                <div id="scores-table-container-${year}" class="table-container rounded-lg shadow-lg bg-[#151c2c] border border-gray-700">

                    <table class="w-full text-sm"><thead id="scores-table-head-${year}"></thead><tbody id="scores-table-body-${year}"></tbody></table>

                </div>

                <div id="charts-container-${year}"></div>

            `;

            const head = document.getElementById(`scores-table-head-${year}`);

            const body = document.getElementById(`scores-table-body-${year}`);

            const chartsContainer = document.getElementById(`charts-container-${year}`);

            head.innerHTML = ''; body.innerHTML = ''; chartsContainer.innerHTML = '';

            

            const headerRow = document.createElement('tr');

            headerRow.innerHTML = `<th>Wk</th>` + owners.map(o => {

                const teamFullName = abbrMap[o.teamAbbr] || o.teamAbbr;

                return `<th>${o.name}<br><span class="font-normal text-gray-400 text-xs">${teamFullName}</span></th>`;

            }).join('');

            head.appendChild(headerRow);

            

            let sectionRecords = owners.map(() => ({w: 0, l: 0, t: 0, diff: 0}));

            let sectionPointDiffs = owners.map(() => [0]);

            const subtotalWeeks = [6, 12, 18];

            const today = new Date();

            today.setHours(0,0,0,0);

            

            for (let week = 1; week <= WEEKS; week++) {

                const weekRow = document.createElement('tr');

                let rowContentHTML = `<td class="font-bold text-gray-300">${week}</td>`;

                const gamesThisWeek = allGamesByWeek.get(week) || [];

                owners.forEach((owner, ownerIndex) => {

                    const game = gamesThisWeek.find(g => g.home_team === owner.teamAbbr || g.away_team === owner.teamAbbr);

                    if (game && game.home_score !== null && game.away_score !== null && !isNaN(game.home_score) && !isNaN(game.away_score)) {

                        const isAway = game.away_team === owner.teamAbbr;

                        const myTeam = { abbr: owner.teamAbbr, score: isAway ? game.away_score : game.home_score };

                        const opponentAbbr = isAway ? game.home_team : game.away_team;

                        const opponent = { abbr: opponentAbbr, score: isAway ? game.home_score : game.away_score };

                        const pointDiff = myTeam.score - opponent.score;

                        sectionRecords[ownerIndex].diff += pointDiff;

                        const lastCumulative = sectionPointDiffs[ownerIndex][sectionPointDiffs[ownerIndex].length - 1];

                        sectionPointDiffs[ownerIndex].push(lastCumulative + pointDiff);

                        let resultClass = 'tie';

                        if (pointDiff > 0) { resultClass = 'win'; sectionRecords[ownerIndex].w++; }

                        else if (pointDiff < 0) { resultClass = 'loss'; sectionRecords[ownerIndex].l++; }

                        else { sectionRecords[ownerIndex].t++; }

                        rowContentHTML += `<td class="${resultClass}"><div class="font-semibold">${myTeam.score} - ${opponent.score}</div><div class="text-xs text-gray-400">${isAway ? '@' : 'vs'} ${opponent.abbr}</div></td>`;

                    } else {

                        const lastCumulative = sectionPointDiffs[ownerIndex][sectionPointDiffs[ownerIndex].length - 1];

                        sectionPointDiffs[ownerIndex].push(lastCumulative);

                        if (game) {

                             const isAway = game.away_team === owner.teamAbbr;

                             const opponentAbbr = isAway ? game.home_team : game.away_team;

                             rowContentHTML += `<td>${isAway ? '@' : 'vs'} ${opponentAbbr}</td>`;

                        } else {

                             const weekDate = weekDateMap[week] ? new Date(weekDateMap[week]) : null;

                             const isPastWeek = data.isLive ? (weekDate && weekDate < today) : (week <= currentWeek);

                             rowContentHTML += isPastWeek ? `<td class="bye">BYE</td>` : `<td>—</td>`;

                        }

                    }

                });

                weekRow.innerHTML = rowContentHTML;

                body.appendChild(weekRow);


                if (subtotalWeeks.includes(week)) { 

                    const sectionIndex = subtotalWeeks.indexOf(week);

                    const startWeek = (sectionIndex === 0) ? 1 : subtotalWeeks[sectionIndex - 1] + 1;

                    

                    const isSectionComplete = data.isLive ? (weekDateMap[week] && new Date(weekDateMap[week]) < today) : (currentWeek >= week);


                    if (isSectionComplete) {

                        let winnerIndex = -1;

                        if (games.some(g => g.week >= startWeek && g.week <= week)) {

                            let maxWins = -1;

                            sectionRecords.forEach(r => { if (r.w > maxWins) maxWins = r.w; });

                            

                            const potentialWinners = sectionRecords

                                .map((r, i) => ({...r, index: i}))

                                .filter(r => r.w === maxWins && r.w > r.l);

                            

                            if (potentialWinners.length === 1) {

                                winnerIndex = potentialWinners[0].index;

                            } else if (potentialWinners.length > 1) {

                                let maxDiff = -Infinity;

                                let winner = null;

                                potentialWinners.forEach(p => { 

                                    if (p.diff > maxDiff) { 

                                        maxDiff = p.diff; 

                                        winner = p;

                                    }

                                });

                                if(winner) winnerIndex = winner.index;

                            }

                        }


                        const subtotalRow = document.createElement('tr');

                        subtotalRow.className = 'subtotal-row'; 

                        let subtotalHtml = `<td>Rec (Wk ${startWeek}-${week})</td>`;

                        sectionRecords.forEach((record, i) => {

                            const highlightClass = (i === winnerIndex) ? 'section-winner' : '';

                            subtotalHtml += `<td class="${highlightClass}">${record.w}-${record.l}${record.t > 0 ? `-${record.t}` : ''}</td>`;

                        });

                        subtotalRow.innerHTML = subtotalHtml;

                        body.appendChild(subtotalRow);


                        const chartLabels = Array.from({length: (week - startWeek + 1)}, (_, i) => `Wk ${startWeek + i}`);

                        chartLabels.unshift(`Start`);

                        const chartDatasets = owners.map((owner, i) => ({

                            label: owner.teamAbbr, data: sectionPointDiffs[i],

                            borderColor: teamLineColors[owner.teamAbbr] || '#888',

                            pointBackgroundColor: teamMarkerColors[owner.teamAbbr] || '#888',

                            pointBorderColor: teamMarkerColors[owner.teamAbbr] || '#888',

                            pointRadius: 4, tension: 0.1, fill: false

                        }));

                        const chartTitle = `Cumulative Point Differential: Weeks ${startWeek}-${week}`;

                        const canvasId = `chart-${year}-section-${sectionIndex}`;

                        const legendId = `legend-${year}-section-${sectionIndex}`;

                        chartsContainer.innerHTML += `

                            <div class="chart-wrapper">

                                <h3 class="text-xl font-bold text-center mb-4">${chartTitle}</h3>

                                <div class="chart-container"><canvas id="${canvasId}"></canvas></div>

                                <div id="${legendId}" class="custom-legend"></div>

                            </div>`;

                        renderChart(canvasId, legendId, chartLabels, chartDatasets);

                    }

                    

                    sectionRecords = owners.map(() => ({w: 0, l: 0, t: 0, diff: 0}));

                    sectionPointDiffs = owners.map(() => [0]);

                }

            }

        }


        function renderChart(canvasId, legendId, labels, datasets) {

            const whiteBackgroundPlugin = {

                id: 'customCanvasBackgroundColor',

                beforeDraw: (chart) => {

                    const { ctx } = chart;

                    ctx.save();

                    ctx.globalCompositeOperation = 'destination-over';

                    ctx.fillStyle = '#ffffff';

                    ctx.fillRect(0, 0, chart.width, chart.height);

                    ctx.restore();

                }
